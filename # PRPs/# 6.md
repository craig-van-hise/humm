
## 1. Immediate Guide Tone & Drone Stop (`useAudioEngine.ts`)

To ensure the guide tone stops instantly when tapping the main circle—instead of playing out its remaining duration—we add a `stopAllAudio()` method that immediately triggers note releases on `Tone.Synth` and halts the drone oscillators.

```typescript
// hooks/useAudioEngine.ts
import { useRef, useEffect, useCallback } from 'react';
import * as Tone from 'tone';

interface AudioEngineProps {
  rootNote: string;
  droneOctaveOffset: number;
  droneRootVol: number;
  droneFifthVol: number;
  guideVol: number;
}

export function useAudioEngine({
  rootNote,
  droneOctaveOffset = -1,
  droneRootVol,
  droneFifthVol,
  guideVol,
}: AudioEngineProps) {
  const isEngineInit = useRef(false);
  const guideSynthRef = useRef<Tone.Synth | null>(null);

  const droneRootOscRef = useRef<Tone.OmniOscillator | null>(null);
  const droneFifthOscRef = useRef<Tone.OmniOscillator | null>(null);

  const droneRootGainRef = useRef<Tone.Gain | null>(null);
  const droneFifthGainRef = useRef<Tone.Gain | null>(null);
  const guideGainRef = useRef<Tone.Gain | null>(null);

  // Initialize Engine
  const initEngine = useCallback(async () => {
    if (isEngineInit.current) return;

    await Tone.start();

    droneRootGainRef.current = new Tone.Gain(Tone.gainToDb(droneRootVol)).toDestination();
    droneFifthGainRef.current = new Tone.Gain(Tone.gainToDb(droneFifthVol)).toDestination();
    guideGainRef.current = new Tone.Gain(Tone.gainToDb(guideVol)).toDestination();

    const droneRootFilter = new Tone.Filter(500, "lowpass").connect(droneRootGainRef.current);
    const droneFifthFilter = new Tone.Filter(600, "lowpass").connect(droneFifthGainRef.current);

    droneRootOscRef.current = new Tone.OmniOscillator({ type: "sawtooth" }).connect(droneRootFilter);
    droneFifthOscRef.current = new Tone.OmniOscillator({ type: "triangle" }).connect(droneFifthFilter);

    guideSynthRef.current = new Tone.Synth({
      oscillator: { type: "triangle" },
      envelope: {
        attack: 0.08,
        decay: 0.1,
        sustain: 0.85,
        release: 0.15,
      },
    }).connect(guideGainRef.current);

    isEngineInit.current = true;
  }, [droneRootVol, droneFifthVol, guideVol]);

  // Start Drone
  const startDrone = useCallback(async () => {
    await initEngine();

    const rootMidi = Tone.Frequency(rootNote).toMidi();
    const droneRootMidi = rootMidi + (droneOctaveOffset * 12);
    const droneFifthMidi = droneRootMidi + 7;

    if (droneRootOscRef.current && droneFifthOscRef.current) {
      droneRootOscRef.current.frequency.setValueAtTime(Tone.Frequency(droneRootMidi, "midi").toFrequency(), Tone.now());
      droneFifthOscRef.current.frequency.setValueAtTime(Tone.Frequency(droneFifthMidi, "midi").toFrequency(), Tone.now());

      droneRootOscRef.current.start();
      droneFifthOscRef.current.start();
    }
  }, [initEngine, rootNote, droneOctaveOffset]);

  // Trigger Guide Pitch
  const playGuidePitch = useCallback((pitchFreq: number, durationSec: number) => {
    if (guideSynthRef.current && isEngineInit.current) {
      guideSynthRef.current.triggerAttackRelease(pitchFreq, durationSec, Tone.now());
    }
  }, []);

  // INSTANT STOP: Cuts off both Drone & Guide Tone immediately
  const stopAllAudio = useCallback(() => {
    // 1. Instantly release guide tone envelope
    if (guideSynthRef.current) {
      guideSynthRef.current.triggerRelease(Tone.now());
    }

    // 2. Stop drone oscillators
    if (droneRootOscRef.current && droneFifthOscRef.current) {
      droneRootOscRef.current.stop();
      droneFifthOscRef.current.stop();
    }
  }, []);

  // Dynamic Volume Updates
  useEffect(() => {
    if (droneRootGainRef.current) droneRootGainRef.current.gain.rampTo(droneRootVol, 0.05);
    if (droneFifthGainRef.current) droneFifthGainRef.current.gain.rampTo(droneFifthVol, 0.05);
    if (guideGainRef.current) guideGainRef.current.gain.rampTo(guideVol, 0.05);
  }, [droneRootVol, droneFifthVol, guideVol]);

  return {
    startDrone,
    playGuidePitch,
    stopAllAudio, // Call this on session stop / circle tap
    initEngine,
  };
}

```

## 2. "Custom" Mode Card in Pacing Protocols (`PacingModeSelector.tsx`)

"Custom Timing" is integrated directly as a selectable card inside the **Breath Pacing Protocol** section alongside Vagal Calm, Focus/Theta, and Sinus NO Recharge.

* Tapping the **Custom card** selects Custom timing.
* Clicking the **⚙️ Edit icon** inside the Custom card opens the customization modal.

```tsx
// components/PacingModeSelector.tsx
import React from 'react';
import { Sliders, Wind } from 'lucide-react';
import { TimingSettings } from './TimingSettingsModal';

export type PacingPresetId = 'vagal-calm' | 'focus-theta' | 'sinus-recharge' | 'custom';

export interface PacingModeSelectorProps {
  selectedMode: PacingPresetId;
  onSelectMode: (mode: PacingPresetId) => void;
  customSettings: TimingSettings;
  onOpenCustomModal: () => void;
  sequenceNoteCount: number;
}

export const PacingModeSelector = ({
  selectedMode,
  onSelectMode,
  customSettings,
  onOpenCustomModal,
  sequenceNoteCount,
}: PacingModeSelectorProps) => {
  const PRESETS: { id: PacingPresetId; name: string; subtitle: string; desc: string }[] = [
    {
      id: 'vagal-calm',
      name: 'Vagal Calm',
      subtitle: 'Extended Exhalation',
      desc: '4s / 10s / 4s',
    },
    {
      id: 'focus-theta',
      name: 'Focus / Theta',
      subtitle: 'Shorter Bursts',
      desc: '3s / 6s / 2s',
    },
    {
      id: 'sinus-recharge',
      name: 'Sinus NO Recharge',
      subtitle: 'NO Regeneration',
      desc: '4s / 10s / 180s',
    },
  ];

  // Calculate display string for custom card
  const customHumDisplay =
    customSettings.mode === 'fixed-note'
      ? `${customSettings.noteSec}s/note`
      : `${customSettings.totalHumSec}s total`;

  return (
    <div className="space-y-2">
      <div className="flex items-center gap-1.5 border-b border-slate-800 pb-2">
        <Wind className="w-3.5 h-3.5 text-cyan-400" />
        <h3 className="text-xs font-semibold uppercase tracking-wider text-slate-400">
          Breath Pacing Protocol
        </h3>
      </div>

      {/* 2x2 Grid of Pacing Mode Cards */}
      <div className="grid grid-cols-2 gap-2">
        {/* Preset Cards */}
        {PRESETS.map((preset) => {
          const isSelected = selectedMode === preset.id;

          return (
            <button
              key={preset.id}
              onClick={() => onSelectMode(preset.id)}
              className={`p-3 rounded-xl border text-left transition-all relative flex flex-col justify-between ${
                isSelected
                  ? 'bg-slate-800 border-cyan-400 shadow-[0_0_12px_rgba(34,211,238,0.2)]'
                  : 'bg-slate-900 border-slate-800 hover:border-slate-700'
              }`}
            >
              <div>
                <span className="text-xs font-bold text-white block">
                  {preset.name}
                </span>
                <span className="text-[10px] text-slate-400 block mb-2">
                  {preset.subtitle}
                </span>
              </div>

              <span className="text-[11px] font-mono font-semibold text-cyan-400">
                {preset.desc}
              </span>
            </button>
          );
        })}

        {/* 4th Card: Dedicated "Custom Timing" Mode */}
        <div
          onClick={() => onSelectMode('custom')}
          className={`p-3 rounded-xl border text-left transition-all relative cursor-pointer flex flex-col justify-between ${
            selectedMode === 'custom'
              ? 'bg-slate-800 border-amber-400 shadow-[0_0_12px_rgba(251,191,36,0.2)]'
              : 'bg-slate-900 border-slate-800 hover:border-slate-700'
          }`}
        >
          <div className="flex justify-between items-start">
            <div>
              <span className="text-xs font-bold text-white block">
                Custom Timing
              </span>
              <span className="text-[10px] text-amber-400 font-medium block">
                User Defined
              </span>
            </div>

            {/* Gear/Edit Icon to open timing modal */}
            <button
              onClick={(e) => {
                e.stopPropagation(); // Prevents double toggle
                onSelectMode('custom');
                onOpenCustomModal();
              }}
              className="p-1 rounded-md bg-slate-950/80 text-amber-400 hover:bg-amber-400 hover:text-slate-950 transition-colors"
              title="Configure custom timings"
            >
              <Sliders className="w-3.5 h-3.5" />
            </button>
          </div>

          <div className="text-[11px] font-mono font-semibold text-amber-300 mt-2">
            {customSettings.inhaleSec}s / {customHumDisplay} / {customSettings.restSec}s
          </div>
        </div>
      </div>
    </div>
  );
};
```

## 3. Main App State Wiring (`App.tsx`)

Connecting the session toggle to `stopAllAudio()` and deriving active phase durations based on the selected mode:

```tsx
// App.tsx
import React, { useState } from 'react';
import { usePersistentState } from './hooks/usePersistentState';
import { useAudioEngine } from './hooks/useAudioEngine';
import { PacingModeSelector, PacingPresetId } from './components/PacingModeSelector';
import { TimingSettingsModal, TimingSettings } from './components/TimingSettingsModal';

export default function App() {
  const [selectedPacingMode, setSelectedPacingMode] = usePersistentState<PacingPresetId>(
    'pacingMode',
    'vagal-calm'
  );

  const [customTiming, setCustomTiming] = usePersistentState<TimingSettings>('customTiming', {
    mode: 'fixed-note',
    inhaleSec: 4,
    totalHumSec: 10,
    noteSec: 2.5,
    restSec: 4,
  });

  const [isTimingModalOpen, setIsTimingModalOpen] = useState(false);
  const [isSessionActive, setIsSessionActive] = useState(false);

  // Audio Engine Hook
  const audioEngine = useAudioEngine({
    rootNote: 'C3',
    droneOctaveOffset: -1,
    droneRootVol: 0.6,
    droneFifthVol: 0.35,
    guideVol: 0.9,
  });

  // Toggle Session Handler
  const handleToggleSession = () => {
    if (isSessionActive) {
      // STOP SESSION: Cuts off drone & guide tone instantly
      audioEngine.stopAllAudio();
      setIsSessionActive(false);
    } else {
      // START SESSION
      audioEngine.startDrone();
      setIsSessionActive(true);
    }
  };

  // Derive active timing based on selected mode card
  const getActiveTimings = () => {
    switch (selectedPacingMode) {
      case 'vagal-calm':
        return { inhaleSec: 4, humSec: 10, restSec: 4, noteDurationSec: 2.0 };
      case 'focus-theta':
        return { inhaleSec: 3, humSec: 6, restSec: 2, noteDurationSec: 1.2 };
      case 'sinus-recharge':
        return { inhaleSec: 4, humSec: 10, restSec: 180, noteDurationSec: 2.0 };
      case 'custom':
        const noteCount = 5; // Replace with activeSequence.degrees.length
        const humSec =
          customTiming.mode === 'fixed-note'
            ? customTiming.noteSec * noteCount
            : customTiming.totalHumSec;
        const noteDurationSec =
          customTiming.mode === 'fixed-note'
            ? customTiming.noteSec
            : customTiming.totalHumSec / Math.max(1, noteCount);
        return {
          inhaleSec: customTiming.inhaleSec,
          humSec,
          restSec: customTiming.restSec,
          noteDurationSec,
        };
    }
  };

  return (
    <div className="min-h-screen bg-slate-950 text-white p-4 max-w-md mx-auto space-y-4">
      {/* Pacing Mode Cards */}
      <PacingModeSelector
        selectedMode={selectedPacingMode}
        onSelectMode={setSelectedPacingMode}
        customSettings={customTiming}
        onOpenCustomModal={() => setIsTimingModalOpen(true)}
        sequenceNoteCount={5}
      />

      {/* Custom Timing Config Drawer */}
      <TimingSettingsModal
        isOpen={isTimingModalOpen}
        onClose={() => setIsTimingModalOpen(false)}
        settings={customTiming}
        onUpdate={setCustomTiming}
        sequenceNoteCount={5}
      />
    </div>
  );
}
```
